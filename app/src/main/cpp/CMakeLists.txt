cmake_minimum_required(VERSION 3.22.1)
project(nemotron_native LANGUAGES CXX)
add_library(nemotron_native SHARED nemotron_mel.cpp nemotron_cache.cpp nemotron_session.cpp nemotron_pipeline.cpp)
target_compile_features(nemotron_native PRIVATE cxx_std_17)
target_compile_options(nemotron_native PRIVATE -O3 -Wall -Wextra -Werror)
target_link_options(nemotron_native PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
target_include_directories(nemotron_native PRIVATE third_party/onnxruntime)
target_link_libraries(nemotron_native PRIVATE dl)
if(NATIVE_ASR_C_API)
    target_include_directories(nemotron_native PRIVATE third_party/sherpa)
    add_library(sherpa_c_api SHARED IMPORTED)
    set_target_properties(sherpa_c_api PROPERTIES IMPORTED_LOCATION
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.local/native-runtime/probe-jni/${ANDROID_ABI}/libsherpa-onnx-c-api.so")
    target_link_libraries(nemotron_native PRIVATE sherpa_c_api)
endif()
if(NATIVE_ASR_PROBE)
    target_sources(nemotron_native PRIVATE ../../nativeAsrProbe/cpp/native_asr_probe.cpp)
endif()
if(NATIVE_ASR_CANDIDATE)
    target_sources(nemotron_native PRIVATE native_asr_session.cpp)
endif()
if(SILERO_GAP_PROBE)
    target_sources(nemotron_native PRIVATE native_vad_probe.cpp)
endif()
